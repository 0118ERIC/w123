鬼抓人遊戲 程式碼

from browser import document, html, timer
import math, time

# ===== 建立畫面 =====
document.clear()
document <= html.H2("Brython 鬼抓人（30 秒生存版｜R 重新開始）")

canvas = html.CANVAS(
width=500,
height=400,
style={
"border": "2px solid black",
"backgroundColor": "#222"
}
)
document <= canvas

ctx = canvas.getContext("2d")

W, H = 500, 400

# ===== 障礙物 =====
obstacles = [
(150, 50, 200, 20),
(150, 330, 200, 20),
(240, 120, 20, 160)
]

# ===== 遊戲狀態 =====
keys = set()
win_time = 30

player = {}
ghost = {}
start_time = 0
game_over = False
message = ""

# ===== 初始化 / 重開 =====
def reset_game():
global player, ghost, start_time, game_over, message
player = {"x": 80, "y": 200, "r": 10, "speed": 4}
ghost = {"x": 420, "y": 200, "r": 12, "speed": 2}
start_time = time.time()
game_over = False
message = ""

reset_game()

# ===== 繪圖工具 =====
def draw_circle(x, y, r, color):
ctx.beginPath()
ctx.arc(x, y, r, 0, math.pi * 2)
ctx.fillStyle = color
ctx.fill()

def draw_obstacles():
ctx.fillStyle = "#888"
for x, y, w, h in obstacles:
ctx.fillRect(x, y, w, h)

# ===== 鍵盤 =====
def keydown(e):
keys.add(e.key)
if e.key.lower() == "r" and game_over:
reset_game()

def keyup(e):
keys.discard(e.key)

document.bind("keydown", keydown)
document.bind("keyup", keyup)

# ===== 玩家 =====
def update_player():
ox, oy = player["x"], player["y"]

if "ArrowUp" in keys:
player["y"] -= player["speed"]
if "ArrowDown" in keys:
player["y"] += player["speed"]
if "ArrowLeft" in keys:
player["x"] -= player["speed"]
if "ArrowRight" in keys:
player["x"] += player["speed"]

player["x"] = max(player["r"], min(W - player["r"], player["x"]))
player["y"] = max(player["r"], min(H - player["r"], player["y"]))

for x, y, w, h in obstacles:
if x < player["x"] < x + w and y < player["y"] < y + h:
player["x"], player["y"] = ox, oy

# ===== 鬼 =====
def update_ghost():
if ghost["x"] < player["x"]:
ghost["x"] += ghost["speed"]
if ghost["x"] > player["x"]:
ghost["x"] -= ghost["speed"]
if ghost["y"] < player["y"]:
ghost["y"] += ghost["speed"]
if ghost["y"] > player["y"]:
ghost["y"] -= ghost["speed"]

# ===== 碰撞 =====
def hit():
dx = player["x"] - ghost["x"]
dy = player["y"] - ghost["y"]
return math.hypot(dx, dy) < player["r"] + ghost["r"]

# ===== 主迴圈 =====
def game_loop():
global game_over, message

ctx.clearRect(0, 0, W, H)

elapsed = int(time.time() - start_time)
remain = max(0, win_time - elapsed)

if not game_over:
update_player()
update_ghost()

draw_obstacles()
draw_circle(player["x"], player["y"], player["r"], "#4af")
draw_circle(ghost["x"], ghost["y"], ghost["r"], "#f44")

# 時間
ctx.fillStyle = "white"
ctx.font = "18px Arial"
ctx.fillText(f"剩餘時間：{remain}s", 10, 25)

# 被抓到
if hit() and not game_over:
game_over = True
message = "嘿嘿抓到你摟 太陽出來瞜"

# 勝利
if remain == 0 and not game_over:
game_over = True
message = "你撐到天亮了！"

# 結束訊息（每一幀都畫 → 不會消失）
if message:
ctx.fillStyle = "yellow"
ctx.font = "22px Arial"
ctx.fillText(message, 40, 210)
ctx.font = "16px Arial"
ctx.fillText("按 R 重新開始", 170, 245)

timer.request_animation_frame(lambda t: game_loop())

# ===== 啟動 =====
game_loop()
